# file: src/voltbill/domain/tariff.py
from dataclasses import dataclass


class NegativeUsageError(ValueError):
    def __init__(self, kwh: int) -> None:
        super().__init__(f"usage must not be negative, got {kwh}")
        self.kwh = kwh


@dataclass(frozen=True)
class Tariff:
    first_block_kwh: int
    first_block_pence: int
    above_pence: int

    def price(self, kwh: int) -> int:
        if kwh < 0:
            raise NegativeUsageError(kwh)
        first = min(kwh, self.first_block_kwh)
        return first * self.first_block_pence + (kwh - first) * self.above_pence
```

from dataclasses import dataclass, replace
from datetime import datetime


class StaleAccountError(Exception):
    def __init__(self, meter: str, expected: int) -> None:
        super().__init__(f"account {meter} changed since version {expected}")
        self.expected = expected


@dataclass(frozen=True)
class Account:
    meter: str
    owed_pence: int = 0
    billed_at: datetime | None = None
    version: int = 1

    def charge(self, pence: int, now: datetime) -> "Account":
        return replace(self, owed_pence=self.owed_pence + pence, billed_at=now,
                       version=self.version + 1)
```

# file: src/voltbill/service_layer/billing.py
import asyncio
from collections.abc import Callable
from contextvars import ContextVar
from datetime import datetime
from typing import Protocol

from voltbill.domain.account import Account
from voltbill.domain.tariff import Tariff

current_tenant: ContextVar[str] = ContextVar("current_tenant")


class UnknownMeterError(Exception):
    def __init__(self, meter: str) -> None:
        super().__init__(f"unknown meter {meter}")
        self.meter = meter


class RatesUnavailableError(Exception):
    def __init__(self, tenant: str) -> None:
        super().__init__(f"no rates for {tenant} in time")
        self.tenant = tenant


class Accounts(Protocol):
    async def get(self, meter: str) -> Account | None:
        """The current tenant's account, or None."""

    async def save(self, account: Account, expected_version: int) -> None:
        """Compare-and-swap on the version."""


class RateService(Protocol):
    """A supplier's service outside this application."""

    async def tariff_for(self, tenant: str) -> Tariff:
        """The tenant's current tariff."""


async def bill_meter(meter: str, kwh: int, accounts: Accounts, rates: RateService,
                     clock: Callable[[], datetime], rate_timeout: float = 2.0) -> int:
    tenant = current_tenant.get()
    account = await accounts.get(meter)
    if account is None:
        raise UnknownMeterError(meter)
    try:
        async with asyncio.timeout(rate_timeout):
            tariff = await rates.tariff_for(tenant)
    except TimeoutError as cause:
        raise RatesUnavailableError(tenant) from cause
    pence = tariff.price(kwh)
    await accounts.save(account.charge(pence, clock()), account.version)
    return pence
```

# file: tests/fakes.py
import asyncio

from voltbill.domain.account import Account, StaleAccountError
from voltbill.service_layer.billing import current_tenant


class FakeAccounts:
    """Keyed by (tenant, meter); checks the version like the real adapter."""

    def __init__(self) -> None:
        self.rows: dict[tuple[str, str], Account] = {}

    def seed(self, tenant: str, account: Account) -> None:
        self.rows[(tenant, account.meter)] = account

    async def get(self, meter: str) -> Account | None:
        found = self.rows.get((current_tenant.get(), meter))
        await asyncio.sleep(0)                  # yield, so two use cases can interleave
        return found

    async def save(self, account: Account, expected_version: int) -> None:
        key = (current_tenant.get(), account.meter)
        if self.rows[key].version != expected_version:
            raise StaleAccountError(account.meter, expected_version)
        self.rows[key] = account

# file: tests/unit/domain/test_tariff.py
import pytest
from hypothesis import example, given
from hypothesis import strategies as st

from voltbill.domain.tariff import NegativeUsageError, Tariff

TARIFF = Tariff(first_block_kwh=100, first_block_pence=30, above_pence=20)
usage = st.integers(min_value=0, max_value=1_000_000)


@pytest.mark.parametrize(
    ("kwh", "pence"),
    [
        pytest.param(0, 0, id="nothing-used"),
        pytest.param(99, 2970, id="one-below-the-block"),
        pytest.param(100, 3000, id="exactly-the-block"),
        pytest.param(101, 3020, id="one-above-the-block"),
    ],
)
def test_price_at_the_block_boundary(kwh: int, pence: int) -> None:
    assert TARIFF.price(kwh) == pence


def test_negative_usage_is_rejected() -> None:
    with pytest.raises(NegativeUsageError) as caught:
        TARIFF.price(-1)
    assert caught.value.kwh == -1


@given(first=usage, extra=usage)
@example(first=99, extra=2)
def test_more_usage_never_costs_less(first: int, extra: int) -> None:
    assert TARIFF.price(first + extra) >= TARIFF.price(first)


@given(kwh=usage)
def test_price_lies_between_the_two_unit_rates(kwh: int) -> None:
    assert 20 * kwh <= TARIFF.price(kwh) <= 30 * kwh

# file: tests/unit/service_layer/test_billing.py
import asyncio
from collections.abc import Iterator
from datetime import datetime, timezone
from unittest.mock import AsyncMock, create_autospec

import pytest

from tests.fakes import FakeAccounts
from voltbill.domain.account import Account, StaleAccountError
from voltbill.domain.tariff import Tariff
from voltbill.service_layer.billing import (
    RateService,
    RatesUnavailableError,
    UnknownMeterError,
    bill_meter,
    current_tenant,
)

NOW = datetime(2026, 4, 1, 0, 0, tzinfo=timezone.utc)
CHEAP = Tariff(100, 10, 5)
DEAR = Tariff(100, 40, 30)


def fixed_clock() -> datetime:
    return NOW


@pytest.fixture
def accounts() -> FakeAccounts:
    fake = FakeAccounts()
    fake.seed("north", Account("m-1"))
    fake.seed("south", Account("m-1"))
    return fake


@pytest.fixture
def rates() -> AsyncMock:
    # A stub: it only feeds a tariff in, so no test asserts its calls.
    stub: AsyncMock = create_autospec(RateService, instance=True)
    stub.tariff_for.side_effect = {"north": CHEAP, "south": DEAR}.__getitem__
    return stub


@pytest.fixture(params=["north", "south"])
def tenant(request: pytest.FixtureRequest) -> Iterator[str]:
    token = current_tenant.set(request.param)
    yield request.param
    current_tenant.reset(token)


async def test_billing_charges_only_the_current_tenant(
    tenant: str, accounts: FakeAccounts, rates: AsyncMock
) -> None:
    other = "south" if tenant == "north" else "north"
    expected = {"north": 100, "south": 400}[tenant]

    assert await bill_meter("m-1", 10, accounts, rates, fixed_clock) == expected

    assert accounts.rows[(tenant, "m-1")] == Account("m-1", expected, NOW, 2)
    assert accounts.rows[(other, "m-1")] == Account("m-1")


async def test_unknown_meter_changes_nothing(
    tenant: str, accounts: FakeAccounts, rates: AsyncMock
) -> None:
    before = dict(accounts.rows)

    with pytest.raises(UnknownMeterError):
        await bill_meter("m-9", 10, accounts, rates, fixed_clock)

    assert accounts.rows == before


async def test_slow_rate_service_changes_nothing(
    tenant: str, accounts: FakeAccounts, rates: AsyncMock
) -> None:
    async def never(_: str) -> Tariff:
        await asyncio.Event().wait()
        return CHEAP

    rates.tariff_for.side_effect = never
    before = dict(accounts.rows)

    with pytest.raises(RatesUnavailableError) as caught:
        await bill_meter("m-1", 10, accounts, rates, fixed_clock, rate_timeout=0)

    assert caught.value.tenant == tenant
    assert accounts.rows == before


async def test_two_bills_at_once_cannot_both_win(
    tenant: str, accounts: FakeAccounts, rates: AsyncMock
) -> None:
    results = await asyncio.gather(
        bill_meter("m-1", 10, accounts, rates, fixed_clock),
        bill_meter("m-1", 10, accounts, rates, fixed_clock),
        return_exceptions=True,
    )

    assert sorted(type(r).__name__ for r in results) == sorted(["int", "StaleAccountError"])
    assert any(isinstance(r, StaleAccountError) and r.expected == 1 for r in results)
    assert accounts.rows[(tenant, "m-1")].version == 2
```
